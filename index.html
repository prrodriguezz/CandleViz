A lógica em JavaScript

function birthdayCakeCandles(candles) {
  const tallest = Math.max(...candles);
  return candles.filter(c => c === tallest).length;
}

// Exemplo:
const heights = [3, 2, 1, 3];
console.log(birthdayCakeCandles(heights)); // → 2
